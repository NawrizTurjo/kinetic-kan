      \pgfmathsetmacro{\hh}{max(-1.1,min(1.1,\b*\bh))}

        \fill[blueacc!85] (\xx-0.08,0) rectangle (\xx+0.08,\hh);
      \else
        \fill[primary] (\xx-0.08,0) rectangle (\xx+0.08,\hh);
      \fi
      \fill[white, draw=ink, line width=0.5pt] (\xx,0) circle (1.6pt);
