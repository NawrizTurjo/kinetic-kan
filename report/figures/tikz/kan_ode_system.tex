% ------------------------------------------------------------------ nodes

\node[hot, minimum width=5.0cm, minimum height=2.1cm] (rk) at (4.9,0)

\node[cool, minimum width=3.1cm, minimum height=1.25cm] (data) at (13.75,2.75)
  {\textbf{Observed data}\\[1pt] $\mathbf{u}(t_k)$, $t_k\in[0,3.5]$};
